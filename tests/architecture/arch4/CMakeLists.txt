cmake_minimum_required(VERSION 3.16)
project(TurtleArch4Tests LANGUAGES CXX)
enable_testing()
get_filename_component(DEFAULT_CORE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CORE_ROOT "${DEFAULT_CORE_ROOT}" CACHE PATH "Turtle source root")
set(TEST_ROOT "${CORE_ROOT}/tests/architecture")
set(SHARED "${CORE_ROOT}/src/shared")
function(arch4_test name)
  add_executable(${name} "${TEST_ROOT}/${name}.cpp" ${ARGN})
  target_compile_features(${name} PRIVATE cxx_std_17)
  target_include_directories(${name} PRIVATE "${SHARED}")
  if(MSVC)
    target_compile_options(${name} PRIVATE /EHsc /UNDEBUG)
  endif()
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES TIMEOUT 90)
endfunction()
arch4_test(Arch4StorageTest)
arch4_test(ThreadPoolBatchTest "${SHARED}/ThreadPool.cpp")
target_compile_definitions(ThreadPoolBatchTest PRIVATE THREADPOOL_STANDALONE_TEST)
file(GLOB DETOUR_SOURCES "${CORE_ROOT}/dep/recastnavigation/Detour/Source/*.cpp")
arch4_test(NavMeshLifetimeTest ${DETOUR_SOURCES})
target_compile_definitions(NavMeshLifetimeTest PRIVATE DT_VIRTUAL_QUERYFILTER=1)
target_include_directories(NavMeshLifetimeTest PRIVATE "${CORE_ROOT}/dep/recastnavigation/Detour/Include")
if(WIN32)
  foreach(name Arch4DiagnosticsTest Arch4ExecutorDiagnosticsTest)
    arch4_test(${name} "${SHARED}/PerfStats.cpp")
    target_compile_definitions(${name} PRIVATE MANTECH_DEV_DIAGNOSTICS)
    target_link_libraries(${name} PRIVATE psapi)
  endforeach()
  arch4_test(Arch4AllocationTest "${CORE_ROOT}/src/mangosd/AllocationProfile.cpp")
  target_link_libraries(Arch4AllocationTest PRIVATE dbghelp)
endif()
if(MSVC)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME Arch4PortRegressions COMMAND "${Python3_EXECUTABLE}"
      "${CORE_ROOT}/tests/architecture/arch4/run_port_regressions.py"
      --output "${CMAKE_CURRENT_BINARY_DIR}/native-port" --cxx "${CMAKE_CXX_COMPILER}")
    set_tests_properties(Arch4PortRegressions PROPERTIES TIMEOUT 180 ENVIRONMENT "PYTHONUTF8=1")
  endif()
endif()
